"""Generate explicit33 frames; old tests22–32 and their helpers stay intact."""
from pathlib import Path
import ast,hashlib,json
ROOT=Path(r'F:\mb-qa-ci33-20261019');records={}
digest=lambda s:hashlib.sha256(s.encode()).hexdigest()
def tests(tree):return {n.name:ast.dump(n,include_attributes=False) for n in tree.body if isinstance(n,ast.FunctionDef) and n.name.startswith('test_')}
def clone(source,target,pairs):
    raw=(ROOT/source).read_bytes().replace(b'\r\n',b'\n').decode();changed=raw
    for a,b in pairs:assert a in changed;(changed:=changed.replace(a,b))
    inverse=changed
    for a,b in reversed(pairs):inverse=inverse.replace(b,a)
    assert inverse==raw and tests(ast.parse(raw))==tests(ast.parse(changed))
    (ROOT/target).write_text(changed,encoding='utf8')
    records[target]={'source':source,'source_normalized_sha256':digest(raw),'current_normalized_sha256':digest(changed),
        'full_inverse_exact':True,'framing_pairs':pairs,'business_functions_exact':sorted(tests(ast.parse(raw)))}
def wrapper(sources,target,imports):
    lines=['"""Current33 fixture binding; imported original function objects retain every assertion."""','import pytest',*imports]
    mapping={};markers=[]
    for source in sources:
        raw=(ROOT/source).read_bytes().replace(b'\r\n',b'\n').decode();tree=ast.parse(raw);names=sorted(tests(tree))
        module=source[:-3].replace('/','.')
        lines.append('from '+module+' import (\n    '+',\n    '.join(names)+',\n)')
        for node in tree.body:
            if isinstance(node,ast.Assign) and any(isinstance(t,ast.Name) and t.id=='pytestmark' for t in node.targets):
                markers.append(ast.get_source_segment(raw,node))
        mapping[source]={'normalized_sha256':digest(raw),'functions':names}
    assert len(set(markers))<=1
    changed='\n\n'.join(lines+sorted(set(markers)))+'\n';ast.parse(changed);(ROOT/target).write_text(changed,encoding='utf8')
    records[target]={'same_original_function_objects':True,'sources':mapping,'current_normalized_sha256':digest(changed)}
