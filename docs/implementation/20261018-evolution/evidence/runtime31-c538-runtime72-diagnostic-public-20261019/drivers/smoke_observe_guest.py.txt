def inspect_owned():
 row=json.loads(subprocess.check_output(['docker','inspect',container]))[0]
 assert row['Id']==container and row['Image']==state['image_id'] and row['Config']['Labels']['metodobase.lab.plan']==state['nonce']

def logs():

 target=root/'startup-application-final-private.log';target.write_bytes(r.stdout+b'\nSTDERR\n'+r.stderr)
 return {'bytes':target.stat().st_size,'sha256':hashlib.sha256(target.read_bytes()).hexdigest()}
def request(path):
 try:
  with urllib.request.urlopen('http://127.0.0.1:18031'+path,timeout=5) as response:
   return {'status':response.status,'body':json.load(response)}
 except urllib.error.HTTPError as error:
  return {'status':error.code,'body':json.loads(error.read())}
 except (urllib.error.URLError,TimeoutError,ConnectionError) as error:
  return {'status':None,'error_class':type(error).__name__}
