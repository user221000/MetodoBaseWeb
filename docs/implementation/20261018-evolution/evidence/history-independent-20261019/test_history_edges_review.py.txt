"""Independent real SQLite page-boundary and live-position behavior in both factories."""
from datetime import datetime

import pytest
from sqlalchemy.orm import Session

from tests.test_personal_history_pagination import history_http, plans, get


@pytest.mark.parametrize('count', [50, 51])
def test_default_frontier_distinguishes_exactly50_from_one_more(history_http, count):
    identities = plans(history_http[1], count)
    first = get(history_http)
    assert first.status_code == 200
    body = first.json()
    assert [row['id'] for row in body['planes']] == identities[::-1][:50]
    if count == 50:
        assert body['next_cursor'] is None
    else:
        assert isinstance(body['next_cursor'], str)
        last = get(history_http, cursor=body['next_cursor'])
        assert last.status_code == 200
        assert [row['id'] for row in last.json()['planes']] == [identities[0]]
        assert last.json()['next_cursor'] is None


def test_older_insert_below_position_is_visible_but_newer_requires_restart(history_http):
    identities = plans(history_http[1], 6, stamp=datetime(2026, 1, 1))
    first = get(history_http, limit=2).json()
    assert [row['id'] for row in first['planes']] == identities[::-1][:2]
    older = plans(history_http[1], 1, stamp=datetime(2025, 1, 1))[0]
    newer = plans(history_http[1], 1, stamp=datetime(2027, 1, 1))[0]
    position = first['next_cursor']
    remaining = []
    for _ in range(4):
        response = get(history_http, limit=2, cursor=position)
        assert response.status_code == 200
        body = response.json()
        remaining.extend(row['id'] for row in body['planes'])
        position = body['next_cursor']
        if position is None:
            break
    assert position is None
    assert remaining == identities[-3::-1] + [older]
    assert newer not in remaining and len(set(remaining)) == len(remaining)
    assert get(history_http).json()['planes'][0]['id'] == newer


def test_deleted_null_cursor_keeps_position_and_new_null_row_requires_restart(history_http):
    from web.database.models import PlanGenerado
    dated = plans(history_http[1], 3)
    undated = plans(history_http[1], 5, stamp=None)
    first = get(history_http, limit=4).json()
    assert [row['id'] for row in first['planes']] == dated[::-1] + [undated[-1]]
    with Session(history_http[1]) as db:
        for identity in [undated[-1], undated[2]]:
            db.get(PlanGenerado, identity).deleted_at = datetime(2026, 1, 3)
        db.commit()
    added = plans(history_http[1], 1, stamp=None)[0]
    rest = get(history_http, limit=4, cursor=first['next_cursor'])
    assert rest.status_code == 200
    assert [row['id'] for row in rest.json()['planes']] == [undated[3], undated[1], undated[0]]
    assert rest.json()['next_cursor'] is None
    assert [row['id'] for row in get(history_http).json()['planes']] == dated[::-1] + [added, undated[3], undated[1], undated[0]]
