def packages():
    result=[]
    for dist in importlib.metadata.distributions():
        rows=[];missing=[]
        for relative in sorted(dist.files or [],key=str):
            if Path(relative).suffix in {'.pyc','.pyo'}:continue
            path=Path(dist.locate_file(relative))
            if path.is_file():rows.append(str(relative)+'\0'+SHA(path.read_bytes()))
            else:missing.append(str(relative))
        result.append({'name':dist.metadata['Name'],'version':dist.version,'installed_files':len(rows),'installed_files_sha256':SHA('\n'.join(rows).encode()),'missing_record_files':missing})
    return sorted(result,key=lambda r:(r['name'].casefold(),r['version']))
